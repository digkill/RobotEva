  def set_touch_callback(self, callback: Optional[Callable[[Tuple[int, int]], Awaitable[None]]]) -> None:
      """Установить callback для обработки касаний экрана"""
      self._touch_callback = callback

  # В display():
  for event in pygame.event.get():
      if event.type in (pygame.MOUSEBUTTONDOWN, pygame.FINGERDOWN):
          # Получаем координаты
          if event.type == pygame.MOUSEBUTTONDOWN:
              touch_x, touch_y = event.pos
          else:  # FINGERDOWN
              touch_x = int(event.x * self._window_size[0])
              touch_y = int(event.y * self._window_size[1])
          
          # Вызываем callback
          if self._touch_callback:
              await self._touch_callback((touch_x, touch_y))

  async def _handle_touch(self, pos: Tuple[int, int]) -> None:
      """Обработка касания экрана - показать случайную забавную анимацию"""
      if not self._touch_enabled:
          return
      
      x, y = pos
      self.logger.info(f"Touch detected at ({x}, {y})")
      
      if not self._touch_animations:
          return
      
      touch_animation = random.choice(self._touch_animations)
      self.logger.info(f"Playing touch animation: {touch_animation}")
      
      self._previous_animation = self.current_animation
      await self.show_animation(touch_animation)
      
      if self._previous_animation:
          await asyncio.sleep(0.3)
          await self.show_animation(self._previous_animation)

АРХИТЕКТУРА:
  
  ┌─────────────────┐
  │   Touchscreen   │ (или мышь)
  └────────┬────────┘
           │ pygame.event
           ▼
  ┌─────────────────┐
  │ SmallSdlDisplay │ display() обрабатывает события
  └────────┬────────┘
           │ callback(pos)
           ▼
  ┌─────────────────┐
  │ DisplayManager  │ _handle_touch(pos)
  └────────┬────────┘
           │
           ├─► Выбрать случайную анимацию
           ├─► Сохранить текущую эмоцию
           ├─► Показать touch анимацию
           └─► Восстановить предыдущую эмоцию

ПОТОК СОБЫТИЙ:
  1. Пользователь касается экрана → pygame.FINGERDOWN
  2. SmallSdlDisplay.display() ловит событие
  3. Извлекаются координаты (x, y)
  4. Вызывается self._touch_callback((x, y))
  5. DisplayManager._handle_touch() получает координаты
  6. Проверяется self._touch_enabled
  7. Выбирается случайная анимация из self._touch_animations
  8. Текущая анимация сохраняется в self._previous_animation
  9. Запускается touch анимация (loop=False)
  10. После завершения восстанавливается предыдущая

КОНФИГУРАЦИЯ:
  • По умолчанию: все 10 анимаций включены
  • Если animations пуст или не указан → используются все
  • Если enabled=false → touch игнорируется
  • Работает только с backend: sdl

ТРЕБОВАНИЯ:
  ✓ pygame (для SDL backend)
  ✓ PIL (для рендеринга)
  ✓ asyncio (асинхронная обработка)
  ✓ Desktop GUI (X11/Wayland)
  ✓ Touchscreen или мышь

ОГРАНИЧЕНИЯ:
  ✗ Не работает с backend: fbdev (нет обработки событий)
  ✗ Не работает через SSH без X11 forwarding
  ✗ Touch события обрабатываются только в display() цикле
