━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. СИСТЕМНЫЕ ПАКЕТЫ:
   sudo apt update
   sudo apt install -y python3-picamera2 libcamera-apps
   sudo apt install -y python3-opencv python3-numpy python3-scipy
   sudo apt install -y python3-pyaudio portaudio19-dev

2. PYTHON ПАКЕТЫ:
   pip3 install -r requirements.txt --break-system-packages

3. С ВИРТУАЛЬНЫМ ОКРУЖЕНИЕМ:
   python3 -m venv --system-site-packages .venv
   source .venv/bin/activate
   pip install -r requirements.txt

4. БЫСТРАЯ УСТАНОВКА:
   bash install_camera_deps.sh

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. КАМЕРА OV5647:
   • Требует python3-picamera2 (системный пакет)
   • Настройки в config.yaml (type: csi, rotation: 180)

2. OPENCV И NUMPY:
   • Используйте системные версии (python3-opencv, python3-numpy)
   • Лучшая совместимость с picamera2

3. АУДИО:
   • pyaudio через apt (python3-pyaudio)
   • pydub не работает с Python 3.13
   • Используйте aplay для воспроизведения

4. PYTHON 3.13:
   • Удалён модуль audioop (pydub не работает)
   • tflite-runtime недоступен (openwakeword работает без него)
